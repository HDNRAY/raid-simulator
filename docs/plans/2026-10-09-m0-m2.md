```ts
// src/render/scaling.ts —— 逻辑分辨率与自适应缩放的唯一实现
export const LOGICAL_W = 960;
export const LOGICAL_H = 540;
export type ScalingMode = 'integer' | 'fit';

export function computeScale(vw: number, vh: number, mode: ScalingMode): number {
  const fit = Math.min(vw / LOGICAL_W, vh / LOGICAL_H);
  return mode === 'integer' ? Math.max(1, Math.floor(fit)) : fit;
}

/** 固定 canvas 内部尺寸，只改 CSS 尺寸（架构规范 §10.2） */
export function applyCanvasSize(canvas: HTMLCanvasElement, stage: HTMLElement, mode: ScalingMode) {
  const scale = computeScale(stage.clientWidth, stage.clientHeight, mode);
  canvas.style.width = `${LOGICAL_W * scale}px`;
  canvas.style.height = `${LOGICAL_H * scale}px`;
}
```

```ts
export class FixedLoop<S, A> {
```

**种子化 RNG**（状态内，C2 约束的关键）

```ts
// src/core/rng.ts —— mulberry32，状态是单个 uint32，可序列化、可哈希
export function nextRandom(state: number): { value: number; state: number } {
  let t = (state + 0x6d2b79f5) >>> 0;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return { value: ((t ^ (t >>> 14)) >>> 0) / 4294967296, state: t };
}
```

**测试**：`tests/unit/loop.test.ts` 断言

1. 连续推进 60 次 16.6667ms → `tick === 60`
2. 一次巨大的 delta（如 5000ms）只补 `MAX_CATCHUP/DT = 5` 次
3. **同一个 seed + 同一串 action，两次运行的 tick 数与最终状态一致**
4. `alpha ∈ [0, 1)`

### M0 完成记录（2026-10-09）

**8 个任务全部完成并实测。** 验证链：`typecheck` + `lint` + `build` +
87 个单测 + CDP 端到端（`pnpm run verify:e2e`）。

实测结果：

| 项 | 结果 |
|---|---|
| tick 速率 | **60.0 ~ 60.4Hz**（151 tick / 2500ms） |
| 模拟位置 | worker 内（不是主线程退回） |
| 缩放几何 | 逻辑 `(60,60)+120x80` → 屏幕 `(80,80)+160x107`，与 1280/960 倍率精确一致 |
| 控制台 | 无 error / exception |
| 单测 | **122 个**（含 29 个时间精度回归、22 个数值纪律） |

落地时发现的、与计划不同的事实（都已处理）：

1. **工具链版本比预期新很多**：`vite@8.3.4`、`vitest@5.0.3`、`eslint@10.12.0`。
2. **TypeScript 用 6.0.3 而不是 7.0.2**：`@typescript-eslint/parser` 的
   peer 是 `>=4.8.4 <6.1.0`，TS 7 是新的 Go 实现，生态尚未跟上。
   为了保住 lint 硬约束而降级；等 typescript-eslint 支持 TS 7 后再升级。
3. **时间表示踩了三轮坑才定下来**（详见架构规范 §12）：
   浮点累积 → 加 1e-9 容差 → 整数纳秒 + 债务累加（**仍然差 1 tick**，
   因为"债务攒够 N tick"与"`ticksAt` 说 N tick"是两套舍入）。
   最终方案：tick 数由真实时间**直接算出**（唯一真相源），
   时间单位取**皮秒**让整数量精确，`ticksAt` 用 round 把精度预算
   花在"整数 tick 正确"上。
4. **主线程退回路径必须逐帧带快照**：worker 路径的
   `SNAPSHOT_INTERVAL_TICKS` 节流是对的（省序列化），
   但主线程没有序列化成本，节流只会让画面 1 秒才更新一次。
5. **lint 规则的 `files` 模式匹配不到 fixture 路径**：
   `src/core/**/*.ts` 匹配不到 `tools/lint-fixtures/src/core/`。
   这个坑很隐蔽 —— 配置看起来对，`calculateConfigForFile` 却显示规则
   全是 `undefined`，测试会**假通过**。fixture 需要单独的显式条目。
6. **DebugPanel 不能静态 import pixi**：否则 pixi 同时被静态与动态导入，
   动态导入失效（Vite 报 `INEFFECTIVE_DYNAMIC_IMPORT`）。
   改为由入口注入构造器。
7. **数值纪律单独立规范**（架构规范 §12）：整数时间、整数生命、
   截止时刻而非剩余时长、Kahan 聚合、`safeDiv` 防 NaN。
   Kahan 不是万能的（灾难性抵消消不掉），这一点也写进规范了。

### M2 — 判定层（4 个任务）

#### M2-1 hit frame 时间轴

**产出**

- `src/core/combat/hit.ts`
- `tests/unit/hit.test.ts`

**要点**

- `HitDef.at` 是**相对技能开始的毫秒偏移**；引擎按 tick 推进，命中落在哪一个 tick
  由 `Math.floor((castStart + at) / DT)` 决定
- 多段攻击（多个 `HitDef`）各自独立结算时间

**测试**：`at: 0`（瞬发）、`at: 750`（落在第 45 tick）、同一技能两个 hit 段的顺序。

#### M2-2 判定旗标：`invulnerable` / `damageReduce`

**产出**

- `src/core/combat/damage.ts`（扩展）
- `tests/unit/judgement.test.ts`

**要点**

- hit 结算顺序：**先查旗标，再算数值**
  1. `invulnerable` → `miss(reason:'invulnerable')`，HP 不变，**但仍发事件**
  2. `damageReduce` → 伤害乘 `(1 - reduce)`，事件里带 `absorbed`
  3. 正常结算
- **闪避成功必须有独立事件**（`EventDef` 的 `miss`）——否则玩家不知道躲掉了

**测试**（T8）

```
GIVEN buff { duration:300, flags:{invulnerable:true} }
WHEN  hit frame 落在 appliedAt+299  → miss('invulnerable')，HP 不变
WHEN  hit frame 落在 appliedAt+300  → 同上（闭区间）
WHEN  hit frame 落在 appliedAt+301  → 正常结算
```

**边界值 299 / 300 / 301 三个都单独断言。** 一帧之差的 bug 只能这样抓。

#### M2-3 属性溯源重算（T3b）

**产出**：`src/core/stats/pipeline.ts` 的脏标记机制 + `tests/unit/recompute.test.ts`

**要点**：这是数据模型 §7.2 规则 6 的验收——**去掉影响属性的 buff 后，
整棵派生树必须重算**。

**测试**

```

T3b 反向：移除 +50 智力 buff → 回到 0.102 / 0.102 / 0.204
T3c 链式：移除后，依赖涌动的第三层效果也回到旧值
```

**判据**：T3b/T3c 通过。**这两条如果红了，说明实现里存在快照，必须修。**
